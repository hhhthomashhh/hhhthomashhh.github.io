function circleRect(x,y,r,o){return x+r>o.x&&x-r<o.x+o.w&&y+r>o.y&&y-r<o.y+o.h}
function pointSegDist(px,py,x1,y1,x2,y2){const vx=x2-x1,vy=y2-y1,wx=px-x1,wy=py-y1,c1=vx*wx+vy*wy;if(c1<=0)return Math.hypot(px-x1,py-y1);const c2=vx*vx+vy*vy;if(c2<=c1)return Math.hypot(px-x2,py-y2);const b=c1/c2;return Math.hypot(px-(x1+b*vx),py-(y1+b*vy))}
function step(dt){if(paused||resultState||!world)return;simTime+=dt;
  // movers
  for(const m of world.movers){const old=m[m.axis],next=old+m.dir*m.speed*dt;if(next<m.min||next>m.max)m.dir*=-1;m[m.axis]+=m.dir*m.speed*dt}
  // movable barrels
  for(const b of world.barrels){if(!b.alive)continue;if(Math.abs(b.vx||0)>.01){b.x+=b.vx*dt;b.vx*=Math.pow(.7,dt);for(const e of world.enemies)if(e.alive&&Math.hypot(e.x-b.x,e.y-b.y)<e.r+b.r+5&&Math.abs(b.vx)>45){explodeBarrel(b,'impact');break}}}
  // hanging fall
  for(const h of world.hangings){if(!h.alive||!h.falling)continue;h.vy+=830*dt;h.y+=h.vy*dt;let hit=false;for(const e of world.enemies){if(e.alive&&rectCircle(h,e)){killEnemy(e,'fall');hit=true}}for(const b of world.barrels){if(b.alive&&rectCircle(h,b)){explodeBarrel(b,'fall');hit=true}}if(h.y+h.h>=595){h.y=595-h.h;h.vy=0;h.falling=false;hit=true}if(hit&&h.kind==='piano')stats.piano=true}
  // pending events
  for(let i=pending.length-1;i>=0;i--){pending[i].t-=dt;if(pending[i].t<=0){const fn=pending[i].fn;pending.splice(i,1);fn()}}
  // bullets
  if(bullet?.alive)stepBullet(bullet,dt,true);

  if(fired)checkResolution(dt);
